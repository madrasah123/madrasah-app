  <script type="module">

    import { getFirestore, collection, addDoc, getDocs, doc, setDoc, getDoc, query, where } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

    const firebaseConfig = {
      apiKey: "AIzaSyBAKlMt30y2fhSweOIk",
      authDomain: "darul-arkam-1317d.firebaseapp.com",
      projectId: "darul-arkam-1317d",
      storageBucket: "darul-arkam-1317d.appspot.com",
      messagingSenderId: "873427230439",
      appId: "1:873427230439:web:4tcce13e6e4b4503b28c42"
    };

    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);

    // Unlocking Teacher Form
    window.toggleTeacherForm = function() {
      const pass = prompt("শিক্ষক পাসওয়ার্ড দিন:");
      if (pass === "1234") {
        document.getElementById('teacher-panel').style.display = 'block';
        alert("শিক্ষক প্যানেল আনলক হয়েছে!");
      } else if(pass !== null) {
        alert("ভুল পাসওয়ার্ড!");
      }
    };

    // Save Data (Teacher Panel)
    window.saveData = async function(type) {
      try {
        if(type === 'fee') {
          const code = document.getElementById('admin-fee-code').value;
          const info = document.getElementById('admin-fee-info').value;
          if(!code || !info) return alert("সব ঘর পূরণ করুন");
          await addDoc(collection(db, "fees"), { code, info, date: new Date().toLocaleDateString('bn-BD') });
          alert("বেতনের তথ্য আপলোড হয়েছে!");
        } else if(type === 'result') {
          const roll = document.getElementById('admin-result-roll').value;
          const info = document.getElementById('admin-result-info').value;
          if(!roll || !info) return alert("সব ঘর পূরণ করুন");
          await addDoc(collection(db, "results"), { roll, info, date: new Date().toLocaleDateString('bn-BD') });
          alert("রেজাল্ট আপলোড হয়েছে!");
        } else if(type === 'notice') {
          const category = document.getElementById('admin-notice-type').value;
          const className = document.getElementById('admin-notice-class').value;
          const subject = document.getElementById('admin-notice-subject').value;
          const title = document.getElementById('admin-notice-title').value;
          const details = document.getElementById('admin-notice-details').value;
          if(!title || !details) return alert("সব ঘর পূরণ করুন");
          await addDoc(collection(db, "feed"), { category, className, subject, title, details, time: "আজ, " + new Date().toLocaleTimeString('bn-BD', {hour: '2-digit', minute:'2-digit'}) });
          alert("নোটিশ/বাড়ির কাজ পাবলিশ হয়েছে!");
          loadFeed();
        } else if(type === 'contact') {
          const principal = document.getElementById('admin-contact-principal').value;
          const office = document.getElementById('admin-contact-office').value;
          if(!principal || !office) return alert("সব নম্বর পূরণ করুন");
          await setDoc(doc(db, "settings", "contact"), { principal, office });
          alert("জরুরি যোগাযোগ নম্বর সফলভাবে আপলোড হয়েছে!");
          loadSettings();
        } else if(type === 'holiday') {
          const eid = document.getElementById('admin-holiday-eid').value;
          const qadr = document.getElementById('admin-holiday-qadr').value;
          if(!eid || !qadr) return alert("সব ছুটির সময় পূরণ করুন");
          await setDoc(doc(db, "settings", "holidays"), { eid, qadr });
          alert("ছুটির সময়সূচি সফলভাবে আপলোড হয়েছে!");
          loadSettings();
        } else if(type === 'addClass') {
          const className = document.getElementById('admin-new-class').value;
          if(!className) return alert("শ্রেণির নাম লিখুন");
          await addDoc(collection(db, "classes"), { name: className });
          alert("নতুন শ্রেণি যুক্ত হয়েছে!");
          document.getElementById('admin-new-class').value = "";
          loadClassesAndSubjects();
        } else if(type === 'addSubject') {
          const subjectName = document.getElementById('admin-new-subject').value;
          if(!subjectName) return alert("বিষয় নাম লিখুন");
          await addDoc(collection(db, "subjects"), { name: subjectName });
          alert("নতুন বিষয় যুক্ত হয়েছে!");
          document.getElementById('admin-new-subject').value = "";
          loadClassesAndSubjects();
        }
      } catch(e) { alert("ত্রুটি: " + e.message); }
    };

    // Load Dynamic Classes and Subjects
    window.loadClassesAndSubjects = async function() {
      try {
        const defaultClasses = ["১ম শ্রেণি", "২য় শ্রেণি", "৩য় শ্রেণি"];
        const defaultSubjects = ["সাধারণ", "কুরআন", "হাদীস", "গণিত", "বাংলা", "ইংরেজি"];

        const classSnap = await getDocs(collection(db, "classes"));
        let classList = [...defaultClasses];
        classSnap.forEach(doc => { classList.push(doc.data().name); });
        classList = [...new Set(classList)];

        const adminNoticeClass = document.getElementById('admin-notice-class');
        const filterClass = document.getElementById('filter-class');
        const msgClass = document.getElementById('msg-class');

        let classOpts = '<option value="সকল শ্রেণি">সকল শ্রেণি</option>';
        let filterOpts = '<option value="সকল শ্রেণীর তথ্য">সকল শ্রেণীর তথ্য</option>';
        let msgOpts = '';

        classList.forEach(c => {
          classOpts += `<option value="${c}">${c}</option>`;
          filterOpts += `<option value="${c}">${c}</option>`;
          msgOpts += `<option value="${c}">${c}</option>`;
        });

        if(adminNoticeClass) adminNoticeClass.innerHTML = classOpts;
        if(filterClass) filterClass.innerHTML = filterOpts;
        if(msgClass) msgClass.innerHTML = msgOpts;

        const subjectSnap = await getDocs(collection(db, "subjects"));
        let subjectList = [...defaultSubjects];
        subjectSnap.forEach(doc => { subjectList.push(doc.data().name); });
        subjectList = [...new Set(subjectList)];

        const adminNoticeSubject = document.getElementById('admin-notice-subject');
        let subjOpts = '';
        subjectList.forEach(s => {
          subjOpts += `<option value="${s}">${s}</option>`;
        });
        if(adminNoticeSubject) adminNoticeSubject.innerHTML = subjOpts;

      } catch(e) {}
    };

    // Load Settings (Contacts and Holidays)
    window.loadSettings = async function() {
      try {
        const contactSnap = await getDoc(doc(db, "settings", "contact"));
        if(contactSnap.exists()) {
          const data = contactSnap.data();
          document.getElementById('contact-display').innerHTML = `
            <b>মুহতামিম/প্রধান শিক্ষক:</b> ${data.principal}<br>
            <b>অফিস কক্ষ:</b> ${data.office}
          `;
          document.getElementById('admin-contact-principal').value = data.principal;
          document.getElementById('admin-contact-office').value = data.office;
        }

        const holidaySnap = await getDoc(doc(db, "settings", "holidays"));
        if(holidaySnap.exists()) {
          const data = holidaySnap.data();
          document.getElementById('holiday-display').innerHTML = `
            <b>ঈদুল ফিতরের ছুটি:</b> ${data.eid}<br>
            <b>শব-ই কদরের ছুটি:</b> ${data.qadr}
          `;
          document.getElementById('admin-holiday-eid').value = data.eid;
          document.getElementById('admin-holiday-qadr').value = data.qadr;
        }
      } catch(e) {}
    };

    // Fetch Fee Info
    window.searchFee = async function() {
      const code = document.getElementById('fee-code-input').value;
      const resDiv = document.getElementById('fee-result');
      if(!code) return alert("কোড দিন");
      resDiv.innerHTML = "খোঁজা হচ্ছে...";
      try {
        const q = query(collection(db, "fees"), where("code", "==", code));
        const snap = await getDocs(q);
        let html = "";
        snap.forEach(doc => { html += `<div class='output-box'><b>আপডেট (${doc.data().date}):</b><br>${doc.data().info}</div>`; });
        resDiv.innerHTML = html || "<div class='output-box'>কোনো তথ্য পাওয়া যায়নি।</div>";
      } catch(e) { resDiv.innerHTML = "ত্রুটি: " + e.message; }
    };

    // Fetch Result Info
    window.searchResult = async function() {
      const roll = document.getElementById('result-roll-input').value;
      const resDiv = document.getElementById('result-display');
      if(!roll) return alert("রোল নম্বর দিন");
      resDiv.innerHTML = "খোঁজা হচ্ছে...";
      try {
        const q = query(collection(db, "results"), where("roll", "==", roll));
        const snap = await getDocs(q);
        let html = "";
        snap.forEach(doc => { html += `<div class='output-box'><b>ফলাফল (${doc.data().date}):</b><br>${doc.data().info}</div>`; });
        resDiv.innerHTML = html || "<div class='output-box'>কোনো রেজাল্ট পাওয়া যায়নি।</div>";
      } catch(e) { resDiv.innerHTML = "ত্রুটি: " + e.message; }
    };

    // Send Message from Parent
    window.sendMessage = async function() {
      const name = document.getElementById('msg-name').value;
      const className = document.getElementById('msg-class').value;
      const text = document.getElementById('msg-text').value;
      if(!name || !text) return alert("নাম এবং মেসেজ লিখুন");
      try {
        await addDoc(collection(db, "messages"), { name, className, text, date: new Date().toLocaleDateString('bn-BD') });
        alert("মেসেজ সফলভাবে পাঠানো হয়েছে!");
        document.getElementById('msg-text').value = "";
        loadMessages();
      } catch(e) { alert("ত্রুটি: " + e.message); }
    };

    // Load Feed (Notices & Homework)
    window.loadFeed = async function() {
      const selectedClass = document.getElementById('filter-class').value;
      const feedDiv = document.getElementById('feed-list');
      feedDiv.innerHTML = "লোড হচ্ছে...";
      try {
        const snap = await getDocs(collection(db, "feed"));
        let html = "";
        snap.forEach(doc => {
          const d = doc.data();
          if(selectedClass === "সকল শ্রেণীর তথ্য" || d.className === selectedClass || d.className === "সকল শ্রেণি") {
            html += `
              <div class="feed-item">
                <div class="badge-container">
                  <span class="badge-hw">${d.category}</span>
                  <span class="badge-class">${d.className}</span>
                  ${d.subject ? `<span class="badge-subject">${d.subject}</span>` : ''}
                </div>
                <div class="feed-title">${d.title}</div>
                <div class="feed-body">${d.details}</div>
                <div class="feed-time">${d.time || 'আজ'}</div>
              </div>
            `;
          }
        });
        feedDiv.innerHTML = html || "কোনো আপডেট নেই।";
      } catch(e) { feedDiv.innerHTML = "ত্রুটি: " + e.message; }
    };

    // Load Sent Messages
    window.loadMessages = async function() {
      const msgDiv = document.getElementById('sent-messages-list');
      try {
        const snap = await getDocs(collection(db, "messages"));
        let html = "";
        snap.forEach(doc => {
          const d = doc.data();
          html += `<div class="contact-box" style="margin-bottom:8px;"><b>${d.name} (${d.className}):</b><br>"${d.text}"</div>`;
        });
        msgDiv.innerHTML = html;
      } catch(e){}
    };

    window.onload = function() {
      loadClassesAndSubjects();
      loadFeed();
      loadMessages();
      loadSettings();
    };
